cmake_minimum_required(VERSION 3.16)
project(fitd_android_hardware_diagnostics LANGUAGES C CXX)

if(NOT SDL2_ROOT OR NOT SDL2_INCLUDE_DIR)
    message(FATAL_ERROR "Pass SDL2_ROOT and SDL2_INCLUDE_DIR, prebuilt for API19 armeabi-v7a")
endif()
set(SDL2_LIBRARY "${SDL2_ROOT}/libSDL2.so")
if(NOT EXISTS "${SDL2_LIBRARY}")
    message(FATAL_ERROR "No SDL2 ARMv7 library at ${SDL2_LIBRARY}")
endif()
add_library(main SHARED diagnostic_gl.cpp)
target_include_directories(main PRIVATE "${SDL2_INCLUDE_DIR}")
target_compile_features(main PRIVATE cxx_std_11)
target_compile_options(main PRIVATE -O2 -DNDEBUG -fvisibility=hidden -fno-strict-aliasing)
target_link_libraries(main PRIVATE "${SDL2_LIBRARY}" GLESv2 EGL log android)
set_target_properties(main PROPERTIES LINK_FLAGS "-Wl,--no-undefined")
